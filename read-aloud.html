(function(){

  const qs = s => document.querySelector(s);

  const qsa = s => Array.from(document.querySelectorAll(s));


  // Elements

  const tabEdit = qs('#tab-edit');

  const tabRead = qs('#tab-read');

  const panelEdit = qs('#panel-edit');

  const panelRead = qs('#panel-read');


  const inputText = qs('#inputText');

  const lineCount = qs('#lineCount');

  const charCount = qs('#charCount');

  const sentencesEl = qs('#sentences');

  const progressEl = qs('#progress');

  const currentIdxEl = qs('#currentIdx');

  const countdownEl = qs('#countdown');


  const voiceSelect = qs('#voiceSelect');

  const refreshBtn = qs('#refreshVoices');

  const testVoiceBtn = qs('#testVoice');

  const rateSel = qs('#rate');

  const pauseSecInput = qs('#pauseSec');

  const fontSizeRange = qs('#fontSize');

  const autoScrollCk = qs('#autoScroll');

  const beepCk = qs('#beep');


  const btnStart = qs('#btnStart');

  const btnPause = qs('#btnPause');

  const btnResume = qs('#btnResume');

  const btnStop = qs('#btnStop');

  const btnPrev = qs('#btnPrev');

  const btnNext = qs('#btnNext');


  const btnSample = qs('#btnSample');

  const btnClear = qs('#btnClear');

  const btnGoRead = qs('#btnGoRead');


  const btnDump = qs('#btnDump');

  const btnCopyCode = qs('#btnCopyCode');

  const sourceBox = qs('#sourceBox');


  // State

  const state = {

    sentences: [],

    idx: 0,

    speaking: false,

    paused: false,

    waiting: false,

    countdownTimer: null,

    countdownLeft: 0,

    currentUtterance: null,

    voices: [],

  };


  // ===== Tab logic =====

  function showPanel(which){

    const isEdit = which === 'edit';

    tabEdit.setAttribute('aria-selected', isEdit);

    tabRead.setAttribute('aria-selected', !isEdit);

    panelEdit.hidden = !isEdit;

    panelRead.hidden = isEdit;

  }

  tabEdit.addEventListener('click', () => showPanel('edit'));

  tabRead.addEventListener('click', () => { showPanel('read'); buildSentences();});


  btnGoRead.addEventListener('click', () => { showPanel('read'); buildSentences(); });


  // ===== Text handling =====

  function splitSentences(raw){

    if(!raw) return [];

    // Normalize newlines

    raw = raw.replace(/\r\n?/g, '\n').trim();

    const result = [];

    let buf = '';

    const isBoundary = (c, i, arr) => {

      // Treat ellipsis ... as one boundary

      if(c === '.' && arr[i+1] === '.' && arr[i+2] === '.') return true;

      return '。！!?？；;….' .includes(c);

    };

    const arr = Array.from(raw);

    for(let i=0;i<arr.length;i++){

      const ch = arr[i];

      if(ch === '\n'){

        if(buf.trim()) { result.push(buf.trim()); }

        buf = '';

        continue;

      }

      buf += ch;

      if(isBoundary(ch, i, arr)){

        // include consecutive trailing quotes/spaces

        while(i+1 < arr.length && /["'”’\)\]\s]/.test(arr[i+1])){ buf += arr[++i]; }

        if(buf.trim()) { result.push(buf.trim()); }

        buf = '';

      }

    }

    if(buf.trim()) result.push(buf.trim());

    // Filter empties and tiny artifacts

    return result.map(s => s.replace(/\s+/g,' ').trim()).filter(s => s.length>0);

  }


  function updateStats(){

    const chars = inputText.value.length;

    const lines = splitSentences(inputText.value).length;

    charCount.textContent = chars;

    lineCount.textContent = lines;

  }


  inputText.addEventListener('input', ()=>{

    updateStats();

    persist();

  });


  // ===== Build sentence list in Read panel =====

  function buildSentences(){

    state.sentences = splitSentences(inputText.value);

    sentencesEl.innerHTML = '';

    state.sentences.forEach((s, i)=>{

      const div = document.createElement('div');

      div.className = 'sentence';

      div.style.fontSize = fontSizeRange.value + 'px';

      div.dataset.idx = String(i);

      div.textContent = s;

      div.title = '点击从第 ' + (i+1) + ' 句开始播放';

      div.addEventListener('click', ()=>{

        stopAll();

        state.idx = i;

        updateProgress();

        startSequence();

      });

      sentencesEl.appendChild(div);

    });

    updateProgress();

  }


  fontSizeRange.addEventListener('input', ()=>{

    qsa('.sentence').forEach(el=> el.style.fontSize = fontSizeRange.value + 'px');

    persist();

  });


  // ===== Voices =====

  function loadVoices(){

    const voices = speechSynthesis.getVoices();

    if(voices && voices.length){

      state.voices = voices;

      voiceSelect.innerHTML = voices.map((v,i)=>{

        const label = `${v.name}  (${v.lang})${v.default?' · 默认':''}`;

        return `<option value="${i}">${label}</option>`;

      }).join('');

      restoreVoiceSelection();

    }

  }

  loadVoices();

  window.speechSynthesis.onvoiceschanged = loadVoices;


  refreshBtn.addEventListener('click', loadVoices);


  testVoiceBtn.addEventListener('click', ()=>{

    const sample = 'This is a voice test. 这是一段音色测试。';

    say(sample);

  });


  function selectedVoice(){

    const idx = parseInt(voiceSelect.value || '0', 10);

    return state.voices[idx] || null;

  }


  // ===== Speech helpers =====

  function say(text, onend){

    if(!text) return;

    cancelSpeechOnly();

    const u = new SpeechSynthesisUtterance(text);

    const v = selectedVoice();

    if(v) u.voice = v;

    u.rate = parseFloat(rateSel.value || '1');

    u.onend = () => { onend && onend(); };

    u.onerror = () => { onend && onend(); };

    state.currentUtterance = u;

    state.speaking = true;

    state.paused = false;

    speechSynthesis.speak(u);

  }


  function cancelSpeechOnly(){

    try{ speechSynthesis.cancel(); } catch(e){}

    state.speaking = false;

  }


  function stopAll(){

    // Cancel timers & speech & reset waiting

    if(state.countdownTimer){ clearInterval(state.countdownTimer); state.countdownTimer=null; }

    cancelSpeechOnly();

    state.waiting = false;

    state.paused = false;

    countdownEl.classList.remove('live');

    countdownEl.textContent = '—';

  }


  function highlight(){

    qsa('.sentence').forEach((el, i)=>{

      el.classList.toggle('active', i===state.idx);

      el.classList.toggle('done', i<state.idx);

    });

    if(autoScrollCk.checked){

      const curr = qs(`.sentence[data-idx="${state.idx}"]`);

      curr && curr.scrollIntoView({behavior:'smooth', block:'center'});

    }

  }


  function updateProgress(){

    progressEl.textContent = `${Math.min(state.idx, state.sentences.length)} / ${state.sentences.length}`;

    currentIdxEl.textContent = state.sentences.length ? (state.idx+1) : '—';

    highlight();

  }


  // Countdown with optional beep

  let audioCtx = null;

  function beep(){

    if(!beepCk.checked) return;

    try{

      audioCtx = audioCtx || new (window.AudioContext||window.webkitAudioContext)();

      const o = audioCtx.createOscillator();

      const g = audioCtx.createGain();

      o.frequency.value = 1200; // Hz

      o.type = 'sine';

      o.connect(g); g.connect(audioCtx.destination);

      g.gain.value = 0.05;

      o.start();

      setTimeout(()=>{ o.stop(); }, 90);

    }catch(e){}

  }


  function startPauseThenNext(){

    state.waiting = true;

    state.speaking = false;

    state.countdownLeft = Math.max(0.5, parseFloat(pauseSecInput.value || '2'));

    countdownEl.classList.add('live');

    countdownEl.textContent = state.countdownLeft.toFixed(1);

    beep();

    if(state.countdownTimer){ clearInterval(state.countdownTimer); }

    const tick = 100;

    state.countdownTimer = setInterval(()=>{

      if(state.paused){ return; }

      state.countdownLeft = Math.max(0, state.countdownLeft - tick/1000);

      countdownEl.textContent = state.countdownLeft.toFixed(1);

      if(Math.abs(state.countdownLeft - Math.round(state.countdownLeft)) < 0.05) beep();

      if(state.countdownLeft <= 0){

        clearInterval(state.countdownTimer); state.countdownTimer=null;

        state.waiting = false;

        countdownEl.classList.remove('live');

        countdownEl.textContent = '—';

        state.idx++;

        updateProgress();

        if(state.idx < state.sentences.length){

          speakCurrent();

        }

      }

    }, tick);

  }


  function speakCurrent(){

    updateProgress();

    const text = state.sentences[state.idx] || '';

    if(!text){ stopAll(); return; }

    say(text, ()=>{

      if(state.paused){ return; }

      startPauseThenNext();

    });

  }


  function startSequence(){

    if(!state.sentences.length){ buildSentences(); }

    if(!state.sentences.length){ alert('请先在“输入文本”页粘贴/输入课文内容'); return; }

    stopAll();

    speakCurrent();

  }


  // ===== Transport controls =====

  btnStart.addEventListener('click', startSequence);

  btnStop.addEventListener('click', ()=>{ stopAll(); state.idx=0; updateProgress(); });

  btnPause.addEventListener('click', ()=>{

    state.paused = true;

    try{ speechSynthesis.pause(); }catch(e){}

  });

  btnResume.addEventListener('click', ()=>{

    state.paused = false;

    try{ if(speechSynthesis.paused) speechSynthesis.resume(); }catch(e){}

  });

  btnPrev.addEventListener('click', ()=>{

    stopAll();

    state.idx = Math.max(0, state.idx-1);

    speakCurrent();

  });

  btnNext.addEventListener('click', ()=>{

    stopAll();

    state.idx = Math.min(state.sentences.length-1, state.idx+1);

    speakCurrent();

  });


  // Keyboard shortcuts

  document.addEventListener('keydown', (e)=>{

    if(e.code === 'Space'){ e.preventDefault(); if(state.paused){ btnResume.click(); } else { btnPause.click(); } }

    if(e.key.toLowerCase()==='j'){ btnPrev.click(); }

    if(e.key.toLowerCase()==='k'){ btnNext.click(); }

  });


  // Persist settings

  function persist(){

    const data = {

      text: inputText.value,

      rate: rateSel.value,

      pause: pauseSecInput.value,

      font: fontSizeRange.value,

      autoScroll: autoScrollCk.checked,

      beep: beepCk.checked,

      voiceName: selectedVoice() ? selectedVoice().name : null

    };

    localStorage.setItem('ra_settings_v1', JSON.stringify(data));

  }


  function restoreVoiceSelection(){

    try{

      const data = JSON.parse(localStorage.getItem('ra_settings_v1')||'{}');

      if(data.voiceName && state.voices.length){

        const idx = state.voices.findIndex(v=> v.name===data.voiceName);

        if(idx>=0){ voiceSelect.value = String(idx); }

      }

    }catch(e){}

  }


  function restore(){

    try{

      const data = JSON.parse(localStorage.getItem('ra_settings_v1')||'{}');

      if(data.text){ inputText.value = data.text; }

      if(data.rate){ rateSel.value = data.rate; }

      if(data.pause){ pauseSecInput.value = data.pause; }

      if(data.font){ fontSizeRange.value = data.font; }

      if(typeof data.autoScroll==='boolean'){ autoScrollCk.checked = data.autoScroll; }

      if(typeof data.beep==='boolean'){ beepCk.checked = data.beep; }

      updateStats();

    }catch(e){}

  }

  restore();


  rateSel.addEventListener('change', persist);

  pauseSecInput.addEventListener('change', persist);

  autoScrollCk.addEventListener('change', persist);

  beepCk.addEventListener('change', persist);

  voiceSelect.addEventListener('change', persist);


  // Sample text

  const sample = `Good morning, everyone. Today we will practice shadow reading. 请同学们注意语音语调与停连。\n\nShadow reading trains your ear, your mouth, and your memory. 它可以让你的语音更自然，更地道。\n\nWhen you hear me read a sentence, wait for the beep, then repeat after me. 当你听到提示音后，再开始跟读。`;


  btnSample.addEventListener('click', ()=>{

    inputText.value = sample;

    updateStats();

    persist();

    buildSentences();

  });

  btnClear.addEventListener('click', ()=>{ inputText.value=''; updateStats(); persist(); sentencesEl.innerHTML=''; updateProgress(); });


  // Dump source

  btnDump.addEventListener('click', ()=>{

    try{

      const full = '<!DOCTYPE html>\n' + document.documentElement.outerHTML;

      sourceBox.value = full;

      sourceBox.focus();

      sourceBox.select();

    }catch(e){ alert('生成失败：' + e.message); }

  });

  btnCopyCode.addEventListener('click', ()=>{

    if(!sourceBox.value){ btnDump.click(); }

    navigator.clipboard.writeText(sourceBox.value).then(()=>{

      btnCopyCode.textContent = '已复制';

      setTimeout(()=> btnCopyCode.textContent = '复制代码', 1200);

    }).catch(()=>{

      sourceBox.select(); document.execCommand('copy');

    });

  });


  // Initial build

  buildSentences();

})();
